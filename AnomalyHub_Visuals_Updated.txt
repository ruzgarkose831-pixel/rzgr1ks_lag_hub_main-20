-- ANTI RAGDOLL (Tools)
-- =====================================================
local antiRagdollEnabled = false
local antiRagdollConn = nil
local antiRagdollCooldown = 0

local function forceResetRagdoll()
    local char = localPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root or hum.Health <= 0 then return end

    pcall(function()
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        root.Velocity = Vector3.zero
        root.RotVelocity = Vector3.zero
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero

        for _, obj in ipairs(char:GetDescendants()) do
            if obj:IsA("Motor6D") then obj.Enabled = true end
            if obj:IsA("Constraint") then obj.Enabled = true end
        end

        if camera then
            camera.CameraSubject = hum
        end

        local PM = localPlayer:FindFirstChild("PlayerScripts") and localPlayer.PlayerScripts:FindFirstChild("PlayerModule")
        if PM then
            local ok, CM = pcall(function()
                return require(PM:FindFirstChild("ControlModule"))
            end)
            if ok and CM and CM.Enable then
                CM:Enable()
            end
        end

        hum.AutoRotate = true
        hum.PlatformStand = false
        hum.Sit = false
    end)
end

local function startAntiRagdoll()
    if antiRagdollConn then return end
    antiRagdollEnabled = true
    antiRagdollConn = RunService.Heartbeat:Connect(function()
        if not antiRagdollEnabled then return end
        local char = localPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end

        local state = hum:GetState()
        local isRagdolled = (state == Enum.HumanoidStateType.Physics or
                             state == Enum.HumanoidStateType.Ragdoll or
                             state == Enum.HumanoidStateType.FallingDown)

        if isRagdolled then
            local now = tick()
            if now - antiRagdollCooldown > 0.15 then
                antiRagdollCooldown = now
                forceResetRagdoll()
            end
        end
    end)
end

local function stopAntiRagdoll()
    antiRagdollEnabled = false
    if antiRagdollConn then
        antiRagdollConn:Disconnect()
        antiRagdollConn = nil
    end
end

local _, AntiRagdollBtn, AntiRagdollBg, AntiRagdollCircle = createToggleRow(toolsPage, "Anti Ragdoll", 2)
AntiRagdollBtn.MouseButton1Click:Connect(function()
    if not antiRagdollEnabled then
        startAntiRagdoll()
        TweenService:Create(AntiRagdollBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(0, 200, 100)}):Play()
        TweenService:Create(AntiRagdollCircle, TweenInfo.new(0.2), {Position = UDim2.new(1, -20, 0.5, -9)}):Play()
    else
        stopAntiRagdoll()
        TweenService:Create(AntiRagdollBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(40, 40, 40)}):Play()
        TweenService:Create(AntiRagdollCircle, TweenInfo.new(0.2), {Position = UDim2.new(0, 2, 0.5, -9)}):Play()
    end
end)

            local ping = stats.Network.ServerStatsItem["Data Ping"]:GetValue()
            StealPing.Text = "PING: " .. math.floor(ping) .. "ms"
        end)
    end)

    -- Drag
    local dragging, dragStart, startPos = false, nil, nil
    StealBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = StealBar.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            StealBar.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

    return StealGui
end

local function stopAutoStealV1()

local function startAutoStealV1()
    stopAutoStealV1()
    local holdDuration = 1.3
    local cooldownAfter = 0.2
    local currentPrompt = nil
    local holdStart = 0

    autoStealV1Conn = RunService.Heartbeat:Connect(function()
        if not autoStealEnabled or autoStealMode ~= "v1" then return end

        local char = localPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        -- En yakın "Steal" prompt bul
        local bestPrompt, bestDist = nil, math.huge
        for _, desc in ipairs(Workspace:GetDescendants()) do
            if desc:IsA("ProximityPrompt") and desc.Name == "Steal" and desc.Enabled then
                local parent = desc.Parent
                local pos = nil
                if parent and parent:IsA("BasePart") then
                    pos = parent.Position
                elseif parent and parent:IsA("Attachment") then
                    pos = parent.WorldPosition
                elseif parent then
                    local p = parent:FindFirstChildWhichIsA("BasePart")
                    if p then pos = p.Position end
                end
                if pos then
                    local d = (hrp.Position - pos).Magnitude
                    if d < bestDist and d <= autoStealRadius then
                        bestDist = d
                        bestPrompt = desc
                    end
                end
            end
        end

        if bestPrompt then
            -- Hold time'ı zorla 1.3 yap
            pcall(function()
                bestPrompt.HoldDuration = holdDuration
            end)

            if not autoStealHolding or currentPrompt ~= bestPrompt then
                autoStealHolding = true
                currentPrompt = bestPrompt
                holdStart = tick()
                autoStealActive = true
                -- Hold başladı simülasyonu
                pcall(function()
                    -- Fire hold began if possible
                    for _, conn in ipairs(getconnections and getconnections(bestPrompt.PromptButtonHoldBegan) or {}) do
                        if conn.Function then pcall(conn.Function) end
                    end
                end)
            end

            local elapsed = tick() - holdStart
            local pct = math.clamp(elapsed / holdDuration, 0, 1)
            updateAutoStealBar(pct)

            if elapsed >= holdDuration then
                -- Trigger
                pcall(function()
                    for _, conn in ipairs(getconnections and getconnections(bestPrompt.Triggered) or {}) do
                        if conn.Function then pcall(conn.Function) end
                    end
                    for _, conn in ipairs(getconnections and getconnections(bestPrompt.PromptButtonHoldEnded) or {}) do
                        if conn.Function then pcall(conn.Function) end
                    end
                end)
                autoStealHolding = false
                currentPrompt = nil
                autoStealActive = false
                updateAutoStealBar(0)
                autoStealLastFinish = tick()
                -- 0.2 sn bekle, sonra yeniden (hala yakındaysa loop devam eder)
            end
        else
            if autoStealHolding then
                autoStealHolding = false
                currentPrompt = nil
                autoStealActive = false
                updateAutoStealBar(0)
            end
        end

        -- Cooldown sonrası tekrar başlayabilir (yukarıdaki logic zaten radius içindeyse yeniden tutar)
        if not autoStealHolding and tick() - autoStealLastFinish < cooldownAfter then
            -- kısa bekleme
        end
    end)
end

-- V2: Dragon Auto Grab logic (sadece steal yöntemi, visual yok)
-- Basitleştirilmiş: aynı radius içinde en yakın steal prompt'u bulup hold + trigger
-- (Tam Dragon scripti çok bağımlı; core hold/trigger + progress bar ile çalışır)
local function stopAutoStealV2()
    if autoStealV2Conn then
        autoStealV2Conn:Disconnect()
        autoStealV2Conn = nil
    end
    autoStealActive = false
    updateAutoStealBar(0)
end

local function startAutoStealV2()
    stopAutoStealV2()
    local HOLD_MIN = 1.3
    local HOLD_MAX = 2.6
    local ENTRY_DELAY = 0.3
    local COOLDOWN = 0.05
    local STEAL_RANGE = 10 -- iç mesafe (prompt'a yaklaşınca trigger)
    local phase = "idle"
    local holdStart = 0
    local currentPrompt = nil

    autoStealV2Conn = RunService.Heartbeat:Connect(function()
        if not autoStealEnabled or autoStealMode ~= "v2" then return end

        local char = localPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        if phase == "idle" or phase == "cooldown" then
            if phase == "cooldown" and tick() - holdStart < COOLDOWN then return end

            -- PRIME_RANGE = autoStealRadius içinde en yakın Steal prompt
            local bestPrompt, bestDist = nil, math.huge
            for _, desc in ipairs(Workspace:GetDescendants()) do
                if desc:IsA("ProximityPrompt") and (desc.Name == "Steal" or string.lower(desc.Name):find("steal")) and desc.Enabled then
                    local parent = desc.Parent
                    local pos = nil
                    if parent and parent:IsA("BasePart") then
                        pos = parent.Position
                    elseif parent and parent:IsA("Attachment") then
                        pos = parent.WorldPosition
                    end
                    if pos then
                        local d = (hrp.Position - pos).Magnitude
                        if d < bestDist and d <= autoStealRadius then
                            bestDist = d
                            bestPrompt = desc
                        end
                    end
                end
            end

            if bestPrompt then
                currentPrompt = bestPrompt
                phase = "holding"
                holdStart = tick()
                autoStealActive = true
                pcall(function()
                    bestPrompt.HoldDuration = HOLD_MIN
                    for _, conn in ipairs(getconnections and getconnections(bestPrompt.PromptButtonHoldBegan) or {}) do
                        if conn.Function then pcall(conn.Function) end
                    end
                end)
            end
        elseif phase == "holding" then
            local elapsed = tick() - holdStart
            local pct = math.clamp(elapsed / HOLD_MAX, 0, 1)
            updateAutoStealBar(pct)

            if not currentPrompt or not currentPrompt.Parent then
                phase = "idle"
                autoStealActive = false
                updateAutoStealBar(0)
                return
            end

            -- Mesafe kontrolü
            local parent = currentPrompt.Parent
            local pos = parent and (parent:IsA("BasePart") and parent.Position or (parent:IsA("Attachment") and parent.WorldPosition))
            local dist = pos and (hrp.Position - pos).Magnitude or 999

            if elapsed >= HOLD_MIN and dist <= STEAL_RANGE then
                task.wait(ENTRY_DELAY)
                pcall(function()
                    for _, conn in ipairs(getconnections and getconnections(currentPrompt.Triggered) or {}) do
                        if conn.Function then pcall(conn.Function) end
                    end
                end)
                phase = "cooldown"
                holdStart = tick()
                autoStealActive = false
                updateAutoStealBar(0)
                currentPrompt = nil
            elseif elapsed >= HOLD_MAX then
                phase = "cooldown"
                holdStart = tick()
                autoStealActive = false
                updateAutoStealBar(0)
                currentPrompt = nil
            end
        end
    end)
end

                end

        end))

        local oldNewIndex

                    spoofedVelocity = value
                    return
                end
            end
            return oldNewIndex(self, key, value)
        end))
    end)

-- --- WALKFLING SİSTEMİ ---
local connections = connections or {}
connections.FreezePlayer = connections.FreezePlayer or {}
local featureStates = featureStates or {}
featureStates.FreezePlayer = featureStates.FreezePlayer or false

local isDropBrActive = false

local function walkfling(enabled)
    local walkflinging = false
    if enabled then
        featureStates.FreezePlayer = true
        local function disablePlayerCollisions()
            local conn = RunService.Stepped:Connect(function()
                if not featureStates.FreezePlayer then
                    conn:Disconnect()
                    return
                end
                local myChar = localPlayer.Character
                if myChar then
                    for _, plr in ipairs(Players:GetPlayers()) do
                        if plr ~= localPlayer and plr.Character then
                            for _, part in ipairs(plr.Character:GetChildren()) do
                                if part:IsA("BasePart") then
                                    part.CanCollide = false
                                end
                            end
                        end
                    end
                end
            end)
            table.insert(connections.FreezePlayer, conn)
        end

        local function stopWalkFling()
            walkflinging = false
        end

        local function startWalkFling()
            local character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
            local humanoid = character:FindFirstChildWhichIsA("Humanoid")
            if humanoid then
                local diedConn = humanoid.Died:Connect(function()
                    stopWalkFling()
                end)
                table.insert(connections.FreezePlayer, diedConn)
            end
            walkflinging = true
            local flingConn = coroutine.create(function()
                local movel = 0.1
                repeat
                    RunService.Heartbeat:Wait()
                    if not featureStates.FreezePlayer then break end

                    character = localPlayer.Character
                    local root = character and character:FindFirstChild("HumanoidRootPart")

                    while not (character and character.Parent and root and root.Parent) do
                        RunService.Heartbeat:Wait()
                        if not featureStates.FreezePlayer then break end
                        character = localPlayer.Character
                        root = character and character:FindFirstChild("HumanoidRootPart")
                    end

                    if not featureStates.FreezePlayer then break end

                    local vel = root.AssemblyLinearVelocity
                    spoofedVelocity = vel

                    local yFlingForce = isDropBrActive and 9e25 or -9e25
                    root.AssemblyLinearVelocity = vel * 100 + Vector3.new(-9e25, yFlingForce, -9e25)
                    
                    RunService.RenderStepped:Wait()
                    if character and character.Parent and root and root.Parent then
                        root.AssemblyLinearVelocity = vel
                    end

                    RunService.Stepped:Wait()
                    if character and character.Parent and root and root.Parent then
                        root.AssemblyLinearVelocity = vel + Vector3.new(0, movel, 0)
                        movel = movel * -1
                    end

                until walkflinging == false or not featureStates.FreezePlayer
            end)
            coroutine.resume(flingConn)
            table.insert(connections.FreezePlayer, flingConn)
        end
        disablePlayerCollisions()
        startWalkFling()
    else
        featureStates.FreezePlayer = false
        walkflinging = false

        local character = localPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if root then
            root.AssemblyLinearVelocity = Vector3.zero
        end
        spoofedVelocity = Vector3.zero

        for _, conn in ipairs(connections.FreezePlayer) do
            if conn then
                if typeof(conn) == "RBXScriptConnection" then
                    conn:Disconnect()
                elseif typeof(conn) == "thread" then
                    task.cancel(conn)
                end
            end
        end
        connections.FreezePlayer = {}
    end
end

local function findMedusaTool()
    local char = localPlayer.Character
    if char then
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") and t.Name:lower():find("medusa") then
                return t
            end
        end
    end
    local bp = localPlayer:FindFirstChild("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and t.Name:lower():find("medusa") then
                return t
            end
        end
    end
    return nil
end

local function counterMedusa()
    local now = tick()
    if now - lastMedusaFire < 1.5 then return end
    lastMedusaFire = now

    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local tool = findMedusaTool()
    if not tool then return end

    if tool.Parent ~= char then
        hum:EquipTool(tool)
    end
    pcall(function() tool:Activate() end)
    for _, v in ipairs(tool:GetDescendants()) do
        if v:IsA("RemoteEvent") then
            pcall(function() v:FireServer() end)
        end
    end
end

local function hookMedusaCharacter(char)
    for _, c in ipairs(medusaConnections) do
        if c and c.Disconnect then
            c:Disconnect()
        end
    end
    table.clear(medusaConnections)

    local head = char:WaitForChild("Head", 5)
    if head then
        table.insert(medusaConnections,
            head:GetPropertyChangedSignal("Anchored"):Connect(function()
                if not medusaCounterEnabled then return end
                if head.Anchored then
                    task.spawn(counterMedusa)
                end
            end)
        )
    end

    local torso = char:FindFirstChild("UpperTorso") or char:WaitForChild("UpperTorso", 3)
    if torso then
        table.insert(medusaConnections,
            torso:GetPropertyChangedSignal("Anchored"):Connect(function()
                if not medusaCounterEnabled then return end
                if torso.Anchored then
                    task.spawn(counterMedusa)
                end
            end)
        )
    end
end

if localPlayer.Character then
    task.spawn(function()
        hookMedusaCharacter(localPlayer.Character)
    end)
end

localPlayer.CharacterAdded:Connect(function(char)
    task.wait(0.15)
    if medusaCounterEnabled then
        hookMedusaCharacter(char)
    end
end)

    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    local bat = findBatTool()
    if not bat then return end

    -- En yakın oyuncuyu bul
    local closest = getClosestPlayer(hrp)
    if not closest or not closest.Character then return end
    local targetHRP = closest.Character:FindFirstChild("HumanoidRootPart")
    if not targetHRP then return end

    -- Bat'ı kuşan
    if bat.Parent ~= char then
        hum:EquipTool(bat)
    end

    -- 0.1 saniye boyunca en yakın kişiye bak
    local lookStart = tick()
    local lookConn
    lookConn = RunService.RenderStepped:Connect(function()
        if not char or not char.Parent or not hrp or not hrp.Parent then
            if lookConn then lookConn:Disconnect() end
            return
        end
        if tick() - lookStart > 0.1 then
            if lookConn then lookConn:Disconnect() end
            return
        end
        local targetPos = targetHRP.Position
        local myPos = hrp.Position
        local lookCFrame = CFrame.lookAt(myPos, Vector3.new(targetPos.X, myPos.Y, targetPos.Z))
        hrp.CFrame = lookCFrame
    end)

    task.delay(0.1, function()
        if lookConn then lookConn:Disconnect() end
        -- Bat'ı kullan
        pcall(function()
            if bat and bat.Parent then
                bat:Activate()
            end
        end)
    end)
end

-- Envanter görünürlük takibi (Backpack + Character tool değişiklikleri)
-- Oyun envanteri gizleyip (tool'lar kaybolur / gizlenir) sonra geri getirince tetikler
local function setupInventoryWatch()
    local function onToolRestored()
        if not batCounterEnabled then return end
        -- Medusa yeni çalıştıysa atla
        if tick() - lastMedusaFire < 2.0 then return end
        task.spawn(doBatCounter)
    end

    local function watchBackpack(bp)
        if not bp then return end
        bp.ChildAdded:Connect(function(child)
            if child:IsA("Tool") and child.Name:lower():find("bat") then
                -- Tool geri geldi → envanter restore olmuş olabilir
                if inventoryWasHidden then
                    inventoryWasHidden = false
                    onToolRestored()
                end
            end
        end)
        bp.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") then
                inventoryWasHidden = true
            end
        end)
    end

    if localPlayer:FindFirstChild("Backpack") then
        watchBackpack(localPlayer.Backpack)
    end
    localPlayer.ChildAdded:Connect(function(child)
        if child.Name == "Backpack" then
            watchBackpack(child)
        end
    end)

    -- Karakter tool'ları da izle (kuşanılıp çıkarılınca)
    local function watchCharacter(char)
        char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") and child.Name:lower():find("bat") then
                if inventoryWasHidden then
                    inventoryWasHidden = false
                    onToolRestored()
                end
            end
        end)
        char.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") then
                inventoryWasHidden = true
            end
        end)
    end

    if localPlayer.Character then
        watchCharacter(localPlayer.Character)
    end
    localPlayer.CharacterAdded:Connect(function(char)
        task.wait(0.1)
        inventoryWasHidden = false
        watchCharacter(char)
    end)
end

setupInventoryWatch()

-- En yakın oyuncuyu bulma
local function getClosestPlayer(rootPart)
    local closestPlr = nil
    local shortestDist = math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local targetHRP = plr.Character.HumanoidRootPart
            local dist = (rootPart.Position - targetHRP.Position).Magnitude
            if dist < shortestDist then
                shortestDist = dist
                closestPlr = plr
            end
        end
    end
    return closestPlr
end

-- --- TP DOWN FONKSİYONU ---
local function TeleportDownToSpawnHeight()
    local char = localPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local targetY = nil

    if localPlayer.RespawnLocation then
        targetY = localPlayer.RespawnLocation.Position.Y
    else
        local spawnPart = Workspace:FindFirstChildWhichIsA("SpawnLocation", true)
        if spawnPart then
            targetY = spawnPart.Position.Y
        end
    end

    local currentPos = hrp.Position

    if targetY then
        hrp.CFrame = CFrame.new(currentPos.X, targetY - 3, currentPos.Z)
    else
        hrp.CFrame = CFrame.new(currentPos.X, currentPos.Y - 3, currentPos.Z)
    end

    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
end

-- --- SOPA KULLANMA DÖNGÜSÜ ---
local lastBatUse = 0
local function EquipAndUseBat()
    local char = localPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local bat = nil
    local backpack = localPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, tool in ipairs(backpack:GetChildren()) do
            if tool:IsA("Tool") and string.lower(tool.Name):find("bat") then
                bat = tool
                break
            end
        end
    end
    if not bat then
        for _, tool in ipairs(char:GetChildren()) do
            if tool:IsA("Tool") and string.lower(tool.Name):find("bat") then
                bat = tool
                break
            end
        end
    end
    if not bat then return end

    if bat.Parent ~= char then
        hum:EquipTool(bat)
    end
    pcall(function()
        bat:Activate()
    end)
end

-- Mirror TP Down takibi
local lastTargetY = nil
local lastTargetPlr = nil

-- --- ANA KONTROL DÖNGÜSÜ (BAT AIMBOT & TP BAT) ---
RunService.RenderStepped:Connect(function()
    local character = localPlayer.Character
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    if buttonStates["Bat Aimbot"] or buttonStates["Tp Bat"] then
        local targetPlr = getClosestPlayer(hrp)
        if targetPlr and targetPlr.Character and targetPlr.Character:FindFirstChild("HumanoidRootPart") then
            local targetHRP = targetPlr.Character.HumanoidRootPart
            local currentTargetY = targetHRP.Position.Y
            
            if mirrorTpDownEnabled and lastTargetPlr == targetPlr and lastTargetY then
                if (lastTargetY - currentTargetY) >= 6 then
                    TeleportDownToSpawnHeight()
                end
            end
            lastTargetY = currentTargetY
            lastTargetPlr = targetPlr

            if buttonStates["Bat Aimbot"] then
                local dist = (targetHRP.Position - hrp.Position).Magnitude
                if dist > 0.5 then
                    local direction = (targetHRP.Position - hrp.Position).Unit
                    hrp.AssemblyLinearVelocity = direction * batSpeedValue
                else
                    hrp.AssemblyLinearVelocity = Vector3.zero
                end
            elseif buttonStates["Tp Bat"] then
                hrp.CFrame = CFrame.new(targetHRP.Position) * hrp.CFrame.Rotation
            end

            local now = tick()
            if now - lastBatUse >= 0.2 then
                lastBatUse = now
                EquipAndUseBat()
            end
        else
            lastTargetY = nil
            lastTargetPlr = nil
        end
    else
        lastTargetY = nil
        lastTargetPlr = nil
    end
end)
